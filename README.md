## 🔬 SQL at Big Data Scale

</div>

```sql
-- ============================================================
--  Advanced Window Functions + CTE on Spark SQL / BigQuery
-- ============================================================
WITH daily_metrics AS (
    SELECT
        user_id,
        event_date,
        COUNT(*)                                          AS events_per_day,
        SUM(revenue)                                      AS daily_revenue,
        LAG(SUM(revenue)) OVER (
            PARTITION BY user_id ORDER BY event_date
        )                                                 AS prev_day_revenue
    FROM   `data_lake.events`
    WHERE  event_date BETWEEN '2024-01-01' AND '2024-12-31'
    GROUP BY 1, 2
),
user_cohort AS (
    SELECT
        user_id,
        event_date,
        daily_revenue,
        SUM(daily_revenue) OVER (
            PARTITION BY user_id ORDER BY event_date
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        )                                                 AS running_total,
        ROUND(
            (daily_revenue - prev_day_revenue)
            / NULLIF(prev_day_revenue, 0) * 100, 2
        )                                                 AS day_over_day_pct,
        NTILE(4) OVER (
            PARTITION BY event_date ORDER BY daily_revenue DESC
        )                                                 AS revenue_quartile
    FROM   daily_metrics
)
SELECT
    event_date,
    COUNT(DISTINCT user_id)                               AS active_users,
    ROUND(AVG(daily_revenue), 2)                          AS avg_revenue,
    PERCENTILE_CONT(0.5) WITHIN GROUP (
        ORDER BY daily_revenue
    )                                                     AS median_revenue,
    SUM(CASE WHEN revenue_quartile = 1 THEN daily_revenue END) AS top_25_revenue
FROM   user_cohort
GROUP BY event_date
ORDER BY event_date;
```
